  it("W5.P concurrent cancels share one native interrupt and release admission together", async () => {
    const agent = createMockAgent();
    let releaseConsumer!: () => void;
    const consumerGate = new Promise<void>(resolve => { releaseConsumer = resolve; });
    injectGeneratorSession(agent, input => {
      async function* messages() {
        const primary = await input[Symbol.asyncIterator]().next();
        yield userEcho(primary.value);
        await consumerGate;
        yield idleMessage();
      }
      return messages();
    });
    const session = agent.sessions["test-session"]!;
    (session as unknown as { clowderCancelQueued: boolean }).clowderCancelQueued = true;
    const releases: Array<() => void> = [];
    const request = vi.fn(async () => {
      await new Promise<void>(resolve => releases.push(resolve));
      return { response: { still_queued: [], cancelled: [] } };
    });
    Object.defineProperty(session.query, "request", { value: request });
    const running = agent.prompt({ sessionId: "test-session", prompt: [{ type: "text", text: "synthetic" }] });
    await waitFor(() => !!session.activeTurn);
    const one = agent.cancel({ sessionId: "test-session" });
    const two = agent.cancel({ sessionId: "test-session" });
    await waitFor(() => request.mock.calls.length > 0);
    try {
      expect(request).toHaveBeenCalledTimes(1);
      await expect(agent.prompt({ sessionId: "test-session", prompt: [{ type: "text", text: "too early" }] })).rejects.toThrow("native_cancel_pending");
      releases[0]!();
      await Promise.all([one, two]);
      expect((session as unknown as { clowderCancelPending: boolean }).clowderCancelPending).toBe(false);
      const later = agent.cancel({ sessionId: "test-session" });
      await waitFor(() => request.mock.calls.length === 2);
      releases[1]!();
      await later;
    } finally {
      for (const release of releases) release();
      await Promise.all([one, two]);
      releaseConsumer();
      await running;
    }
  });
